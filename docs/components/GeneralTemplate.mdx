import {
  Canvas,
  Controls,
  Description,
  Meta,
  Primary,
  Stories,
  Story,
  Title,
} from '@storybook/addon-docs/blocks';

import { PreviewTitle, TitleOfControls } from './Utils.mdx';
import { STORIES_TITLE } from '../utils/utils';

export const GeneralTemplate = ({
  children,
  showStories = true,
  showControls = true,
}) => (
  <section>
    {children && <Meta of={children} />}
    {(children && <Title of={children}>{children?.title}</Title>) || <Title />}
    {(children && <Description of={children}>{children}</Description>) || (
      <Description />
    )}
    <PreviewTitle />
    {(children && (
      <Canvas meta={children} of={children.Default}>
        <Story meta={children} of={children.Default} />
      </Canvas>
    )) || <Primary />}
    {showControls && <TitleOfControls />}
    {(children && showControls && (
      <Controls meta={children} of={children.Default} />
    )) ||
      (showControls && <Controls />)}
    {showStories && <Stories title={STORIES_TITLE} includePrimary={false} />}
  </section>
);
