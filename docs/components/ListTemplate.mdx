import { Title } from '@storybook/addon-docs/blocks';
import { GeneralTemplate } from './GeneralTemplate.mdx';
import { Tabs } from './Tabs.mdx';
import { getFormatName } from '../utils/utils';

export const getPropertyValue = ({item, context, name}) => {
  if(!!item?.default?.title) {
    const contextElement = context[
        getFormatName(
          item?.default.title,
          /(\/)|( )/g,
          "_",
        ).toLocaleLowerCase()
      ];
    if( name === "showControls")  return  contextElement?.showControls;
    if( name === "showStories") return  contextElement?.showStories;

    return true;

}

return true;
};

export const getTabTitle = (item) => item?.default?.title?.split('/').pop();

export const Main = ({ showControls, showStories }) => (
  <GeneralTemplate showControls={showControls} showStories={showStories} />
);

export const Item = ({ children, showControls, showStories }) => (
  <GeneralTemplate showControls={showControls} showStories={showStories}>
    {children}
  </GeneralTemplate>
);

export const ListTemplate = ({
  children,
  main,
  content = [],
  context = {},
  showStories,
  showControls,
}) => {
  return (
    <section>
      {children && (
        <Tabs
          tabs={[<Title />, getTabTitle(children)]}
          content={[
            <Main />,
            <Item showStories={false} showControls={true}>
              {children}
            </Item>,
          ]}
        />
      )}
      {content && main && (
        <Tabs
          tabs={[
            getTabTitle(main),
            ...content.map((item) => getTabTitle(item)),
          ]}
          content={[
            <Main
              showStories={getPropertyValue({
                item: main,
                context,
                name: 'showStories',
              })}
              showControls={getPropertyValue({
                item: main,
                context,
                name: 'showControls',
              })}
            >
              {main}
            </Main>,
            ...content.map((item) => (
              <Item
                showStories={getPropertyValue({
                  item,
                  context,
                  name: 'showStories',
                })}
                showControls={getPropertyValue({
                  item,
                  context,
                  name: 'showControls',
                })}
              >
                {item}
              </Item>
            )),
          ]}
        />
      )}
      {content && !main && (
        <Tabs
          tabs={content.map((item) => getTabTitle(item))}
          content={content.map((item, _, index) => (
            <Item
              showStories={
                !index &&
                getPropertyValue({
                  item,
                  context,
                  name: 'showStories',
                })
              }
              showControls={getPropertyValue({
                item,
                context,
                name: 'showControls',
              })}
            >
              {item}
            </Item>
          ))}
        />
      )}
    </section>
  );
};
