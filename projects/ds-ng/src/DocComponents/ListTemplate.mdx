import { Title } from "@storybook/addon-docs/blocks";
import { GeneralTemplate } from "./GeneralTemplate.mdx";
import { Tabs } from "./Tabs.mdx";
import { getFormatName } from "../lib/utils/doc/utils";

export const getPropertyValue = ({item, context, name}) => {
  if(!!item?.default?.title) {
    const contextElement = context[
        getFormatName(
          item?.default.title,
          /(\/)|( )/g,
          "_",
        ).toLocaleLowerCase()
      ];
    if( name === 'showControls')  return  contextElement?.showControls;
     else if( name === 'showStories') return  contextElement?.showStories;
     return true;
  }

  return true;
};

export const Main = ({ showControls, showStories }) => (
  <section>
    <GeneralTemplate showControls={showControls} showStories={showStories} />
  </section>
);

export const Item = ({ children, showControls, showStories }) => (
  <section>
    <GeneralTemplate showControls={showControls} showStories={showStories}>
      {children}
    </GeneralTemplate>
  </section>
);
/*

<ListTemplate content={[mainStory,itemStory, secondItemStory]}/>
*/
export const ListTemplate = ({
  children,
  main,
  content = [],
  context = { [key]: { showControls, showStories } },
}) => {
  return (
    <section>
      {children && (
        <Tabs
          tabs={[<Title />, <Title of={children}>{children?.title}</Title>]}
          content={[
            <Main
              showControls={getPropertyValue({
                item: children,
                context,
                name: "showControls",
              })}
              showStories={getPropertyValue({
                item: children,
                context,
                name: "showStories",
              })}
            />,
            <Item
              showControls={getPropertyValue({
                item: children,
                context,
                name: "showControls",
              })}
              showStories={getPropertyValue({
                item: children,
                context,
                name: "showStories",
              })}
            >
              {children}
            </Item>,
          ]}
        />
      )}
      {content && main && (
        <Tabs
          tabs={[
            <Title />,
            ...content.map((item) => <Title of={item}>{item?.title}</Title>),
          ]}
          content={[
            <Main
              showControls={getPropertyValue({
                item: children,
                context,
                name: "showControls",
              })}
              showStories={getPropertyValue({
                item: children,
                context,
                name: "showStories",
              })}
            />,
            ...content.map((item) => (
              <Item
                showControls={getPropertyValue({
                  item,
                  context,
                  name: "showControls",
                })}
                showStories={getPropertyValue({
                  item: children,
                  context,
                  name: "showStories",
                })}
              >
                {item}
              </Item>
            )),
          ]}
        />
      )}
      {content && !main && (
        <Tabs
          tabs={content.map((item) => (
            <Title of={item}>{item?.title}</Title>
          ))}
          content={content.map((item) => (
            <Item
              showControls={getPropertyValue({
                item,
                context,
                name: "showControls",
              })}
              showStories={getPropertyValue({
                item: children,
                context,
                name: "showStories",
              })}
            >
              {item}
            </Item>
          ))}
        />
      )}
    </section>
  );
};
