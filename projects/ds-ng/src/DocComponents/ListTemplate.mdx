import { Title } from "@storybook/addon-docs/blocks";
import { GeneralTemplate } from "./GeneralTemplate.mdx";
import { Tabs } from "./Tabs.mdx";

export const List = () => (
  <section>
    <GeneralTemplate />
  </section>
);

export const Item = ({ children }) => (
  <section>
    <GeneralTemplate>{children}</GeneralTemplate>
  </section>
);
/*
<ListTemplate>{itemStory}</ListTemplate>//Two tabs, this takes the main story of the Meta.
<ListTemplate main={mainStory} content={[itemStory, secondItemStory]}/>
<ListTemplate content={[mainStory,itemStory, secondItemStory]}/>
*/
export const ListTemplate = ({ children, main, content = [] }) => {
  return (
    <section>
      {children && (
        <Tabs
          tabs={[<Title />, <Title of={children}>{children?.title}</Title>]}
          content={[<List />, <Item>{children}</Item>]}
        />
      )}
      {content && main && (
        <Tabs
          tabs={[
            <Title />,
            ...content.map((item) => <Title of={item}>{item?.title}</Title>),
          ]}
          content={[<List />, ...content.map((item) => <Item>{item}</Item>)]}
        />
      )}
      {content && !main && (
        <Tabs
          tabs={content.map((item) => (
            <Title of={item}>{item?.title}</Title>
          ))}
          content={content.map((item) => (
            <Item>{item}</Item>
          ))}
        />
      )}
    </section>
  );
};
