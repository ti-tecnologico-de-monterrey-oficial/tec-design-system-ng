import { useState } from "react";
import { TableOfContents } from "@storybook/addon-docs/blocks";

export const ContentTab = ({
  children,
  status,
  showTableOfContents = false,
}) => (
  <div className="sbdocs-wrapper css-3rewwu bmb_doc-tabs-padding-reset">
    {status === "active" && (
      <TableOfContents
        title="Table of contents"
        headingSelector="h2, h3"
        contentsSelector={`#tab_active`}
        disable={!showTableOfContents}
      />
    )}
    <div className="css-x28zkw">
      <div id={`tab_${status}`}>{children}</div>
    </div>
  </div>
);

export const Tabs = ({ tabs = [], content = [], showTableOfContents = false }) => {
  const [selectedTab, setSelectedTab] = useState(0);

return (

<div className="bmb_doc-tabs">
  <ul role="tablist" className="bmb_doc-tabs-header">
    {tabs.map((tab, index) => (
      <li role="tab" key={index} className="bmb_doc-tabs-item">
        <button
          role="button"
          className={`bmb_doc-tabs-button ${selectedTab === index ? "active" : ""}`}
          onClick={() => setSelectedTab(index)}
        >
          {tab}
        </button>
      </li>
    ))}
  </ul>
  <section className="bmb_doc-tabs-content">
    {content.map((item, index) => (
      <div
        role="tabpanel"
        key={index}
        hidden={selectedTab !== index}
        className="bmb_doc-tabs-panel"
      >
        <br />
        <ContentTab
          status={selectedTab === index ? "active" : "inactive"}
          showTableOfContents={showTableOfContents}
        >
          {item}
        </ContentTab>
      </div>
    ))}
  </section>
</div>
); }
